SQL Schema — Jalankan 1x di Supabase SQL Editor 
-- 1. Tabel Users
CREATE TABLE IF NOT EXISTS public.users (
  id TEXT PRIMARY KEY,
  name TEXT, email TEXT UNIQUE,
  password_hash TEXT, role TEXT DEFAULT 'patient',
  condition TEXT, condition_label TEXT,
  recovery_phase TEXT, weight NUMERIC, height NUMERIC,
  age INTEGER, gender TEXT, target_protein NUMERIC,
  target_calories NUMERIC, allergies TEXT,
  symptoms JSONB, created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Tabel Log Makanan & Scan CV
CREATE TABLE IF NOT EXISTS public.meals (
  id TEXT PRIMARY KEY, user_id TEXT,
  meal_type TEXT, name TEXT, calories NUMERIC,
  protein NUMERIC, carbs NUMERIC, fat NUMERIC,
  segments JSONB, timestamp TIMESTAMPTZ DEFAULT NOW()
);

-- Akses publik untuk demo client
ALTER TABLE public.users DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.meals DISABLE ROW LEVEL SECURITY;
